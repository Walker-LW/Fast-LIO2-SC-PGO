cmake_minimum_required(VERSION 3.10)
project(scpgo_dependency_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(GTSAM REQUIRED)
find_package(Ceres REQUIRED)
find_package(OpenCV REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs cv_bridge image_transport pcl_conversions)
add_executable(scpgo_dependency_probe main.cpp)
target_include_directories(scpgo_dependency_probe PRIVATE ${GTSAM_INCLUDE_DIR} ${CERES_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
target_link_libraries(scpgo_dependency_probe PRIVATE gtsam ${CERES_LIBRARIES} ${OpenCV_LIBS} ${PCL_LIBRARIES} ${catkin_LIBRARIES})
